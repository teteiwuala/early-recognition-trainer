Expected file format

One JSON file per session, produced by the sim’s Download session button on the debrief screen. Minimum useful fields; missing fields flag the row as incomplete rather than failing:

{
  "schema_version": 1,
  "pseudonym": "student-042",        // student-entered; never a real name
  "scenario": "sepsis-postop",
  "attempt_label": "attempt 1",       // optional; also inferred from exportedAt order
  "exported_at": "2026-07-27T14:03:00Z",
  "outcome": "escalated_early" | "escalated_late" | "crash",
  "endTime": 214.6,                    // seconds
  "scores": { "ttr": 128.4, "cueLatency": 62.0 },  // cueLatency null = never looked
  "log": [ { "t": 60, "track": "cue|attention|action|reasoning", "key": "...", "label": "...",
             "vitals": {...}, "notice": "...", "interpret": "...", "respond": "...", "skipped": false } ]
}
